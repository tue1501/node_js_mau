-- 4. sanpham_mau_hinhanh
CREATE TABLE `sanpham_mau_hinhanh` (
  `id` int NOT NULL AUTO_INCREMENT,
  `idSanPham` int DEFAULT NULL,
  `tenmau` varchar(255) DEFAULT NULL,
  `hinhanh` varchar(500) DEFAULT NULL,
  `so_luong` int DEFAULT '0',
  PRIMARY KEY (`id`),
  KEY `idSanPham` (`idSanPham`),
  CONSTRAINT `sanpham_mau_hinhanh_ibfk_1` FOREIGN KEY (`idSanPham`) REFERENCES `sanpham` (`idSanPham`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=58 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

-- Trigger cho sanpham_mau_hinhanh
DELIMITER ;;
CREATE TRIGGER `update_tonkho_after_insert` AFTER INSERT ON `sanpham_mau_hinhanh` FOR EACH ROW BEGIN
    UPDATE sanpham
    SET tonkho = (
        SELECT SUM(so_luong)
        FROM sanpham_mau_hinhanh
        WHERE idSanPham = NEW.idSanPham
    )
    WHERE idSanPham = NEW.idSanPham;
END ;;
DELIMITER ;

DELIMITER ;;
CREATE TRIGGER `update_tonkho_after_update` AFTER UPDATE ON `sanpham_mau_hinhanh` FOR EACH ROW BEGIN
    UPDATE sanpham
    SET tonkho = (
        SELECT SUM(so_luong)
        FROM sanpham_mau_hinhanh
        WHERE idSanPham = NEW.idSanPham
    )
    WHERE idSanPham = NEW.idSanPham;
END ;;
DELIMITER ;

DELIMITER ;;
CREATE TRIGGER `update_tonkho_after_delete` AFTER DELETE ON `sanpham_mau_hinhanh` FOR EACH ROW BEGIN
    UPDATE sanpham
    SET tonkho = (
        SELECT SUM(so_luong)
        FROM sanpham_mau_hinhanh
        WHERE idSanPham = OLD.idSanPham
    )
    WHERE idSanPham = OLD.idSanPham;
END ;;
DELIMITER ;
